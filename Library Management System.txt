#include <stdio.h>

int main() {
    // Arrays to store up to 50 book records
    char title[50][50];
    char author[50][50];
    int copies[50];
    
    int count = 0; // Tracks total books currently stored
    int choice, i, search_sno;

    // Menu loop repeats until user chooses Exit (4)
    do {
        printf("\n=====================================\n");
        printf("     LIBRARY MANAGEMENT SYSTEM\n");
        printf("=====================================\n");
        printf("1. Add Book\n");
        printf("2. View All Books\n");
        printf("3. Search Book by S.No.\n");
        printf("4. Exit\n");
        printf("-------------------------------------\n");
        printf("Enter your choice (1-4): ");
        scanf("%d", &choice);

        switch (choice) {
            case 1:
                // Functionality 1: Add book
                if (count >= 50) {
                    printf("\nError: Library storage is full (Max 50 books)!\n");
                } else {
                    printf("\nEnter Book Title: ");
                    scanf(" %[^\n]", title[count]); // Reads string with spaces

                    printf("Enter Author Name: ");
                    scanf(" %[^\n]", author[count]);

                    printf("Enter Number of Copies: ");
                    scanf("%d", &copies[count]);

                    count++; // Increment count; S.No. is equal to new count
                    printf("Book added successfully! Assigned S.No.: %d\n", count);
                }
                break;

            case 2:
                // Functionality 2: View all books
                if (count == 0) {
                    printf("\nNo books found in the library.\n");
                } else {
                    printf("\n-----------------------------------------------------------------\n");
                    printf("%-6s %-25s %-20s %-8s\n", "S.No.", "Title", "Author", "Copies");
                    printf("-----------------------------------------------------------------\n");
                    for (i = 0; i < count; i++) {
                        // S.No. is displayed as (i + 1)
                        printf("%-6d %-25s %-20s %-8d\n", i + 1, title[i], author[i], copies[i]);
                    }
                    printf("-----------------------------------------------------------------\n");
                }
                break;

            case 3:
                // Functionality 3: Search book by S.No.
                if (count == 0) {
                    printf("\nNo books available to search.\n");
                } else {
                    printf("\nEnter S.No. to search: ");
                    scanf("%d", &search_sno);

                    if (search_sno < 1 || search_sno > count) {
                        printf("Record not found! S.No. %d does not exist.\n", search_sno);
                    } else {
                        int idx = search_sno - 1; // Convert 1-based S.No. to 0-based index
                        printf("\n--- Book Details ---\n");
                        printf("S.No.  : %d\n", search_sno);
                        printf("Title  : %s\n", title[idx]);
                        printf("Author : %s\n", author[idx]);
                        printf("Copies : %d\n", copies[idx]);
                    }
                }
                break;

            case 4:
                printf("\nExiting program. Goodbye!\n");
                break;

            default:
                printf("\nInvalid choice! Please enter a number between 1 and 4.\n");
        }
    } while (choice != 4);

    return 0;
}
